{{- /* Shortcuts open catalog cards; doc prefixes identify the project being read. */ -}} {{- $projects := dict -}} {{- range partialCached "project-catalog-data.html" . -}} {{- $project := .project -}} {{- $prefix := "" -}} {{- /* An entry that is part of another project reads that project's docs, which highlight the project's own shortcut. */ -}} {{- if not .partOf -}} {{- with .primary.link -}} {{- if hasPrefix . "/docs/" }}{{ $prefix = replaceRE `^(/docs/[^/]+/).*$` "$1" . }}{{ end -}} {{- end -}} {{- end -}} {{- $catalogLink := printf "/docs/#%s" (replaceRE `[ ()]` "" $project.name) -}} {{- $projects = merge $projects (dict .key (dict "name" $project.name "icon" $project.icon "link" $catalogLink "prefix" $prefix)) -}} {{- end -}} {{- return $projects -}}
