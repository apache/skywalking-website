{{- /* Escapes a value for an iCalendar TEXT field (RFC 5545 §3.3.11).
       Backslash must be escaped first, or it would double-escape the others. */ -}}
{{- $s := . | plainify -}}
{{- $s = replaceRE `\\` `\\\\` $s -}}
{{- $s = replaceRE `;` `\;` $s -}}
{{- $s = replaceRE `,` `\,` $s -}}
{{- $s = replaceRE `\r?\n` `\n` $s -}}
{{- chomp $s -}}
